cmake_minimum_required(VERSION 3.22)
project(AdaptiveMixStereo VERSION 0.8.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED JUCE_DIR)
 message(FATAL_ERROR "Set -DJUCE_DIR to your installed JUCE 7/8 directory")
endif()
add_subdirectory("${JUCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE")
juce_add_plugin(AdaptiveMixStereo COMPANY_NAME "AdaptiveMixLab" BUNDLE_ID "com.adaptivemixlab.stereo" PLUGIN_MANUFACTURER_CODE AdMx PLUGIN_CODE AmS8 FORMATS VST3 PRODUCT_NAME "Adaptive Mix Engine Stereo" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(AdaptiveMixStereo)
target_sources(AdaptiveMixStereo PRIVATE PluginProcessor.cpp)
target_compile_definitions(AdaptiveMixStereo PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(AdaptiveMixStereo PRIVATE juce::juce_audio_utils juce::juce_audio_processors juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
